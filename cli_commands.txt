============================================================
LOFC Recruitment Platform - CLI command reference
Commands to run each phase and (later) the full pipeline.
Updated as we build. Run all commands from the project root.

# ===== Run the WHOLE pipeline (one command) =====
# Populates a fresh database end to end (schema -> ingest -> ... -> shortlists).
# Idempotent. First run downloads ~2.6GB and takes 20-40 min. Then open localhost:8501.
docker compose up -d
docker compose exec app python -m lofc.pipeline


# ----- Phase 0: Scaffold (environment) -----

# Build the app Docker image (installs all Python dependencies)
docker compose build app

# Start the stack (Postgres 16 + app container) in the background
docker compose up -d

# Check both containers are up (db should report "healthy")
docker compose ps

# Confirm config loads from .env (prints settings, open-data mode)
docker compose exec -T app python -c "from lofc.config import settings; print(settings)"

# Apply database migrations (currently the empty baseline schema)
docker compose exec -T app alembic upgrade head

# Run the test suite
docker compose exec -T app pytest -q

# Stop the stack when done (keeps the Postgres data volume)
docker compose down


# ----- Tests & coverage -----

# Run the full test suite
docker compose exec app pytest -q

# Run tests with a coverage report in the terminal (lists untested lines)
docker compose exec app pytest --cov=lofc --cov-report=term-missing

# Generate a browsable HTML coverage report, then open it on your Mac
docker compose exec app pytest --cov=lofc --cov-report=html:data/coverage_html
open data/coverage_html/index.html


# ----- Phase 1: Ingest -----
# Phase 1 lands raw JSON files under data/raw/. It does NOT load Postgres (that is Phase 3).

# Pull all 3 leagues (resumable; skips files already saved)
docker compose exec app python -m lofc.ingest.run

# Pull one league only (2 = Premier League, 11 = La Liga, 12 = Serie A)
docker compose exec app python -m lofc.ingest.run --competition 2

# Quick test: first 5 matches of the Premier League
docker compose exec app python -m lofc.ingest.run --competition 2 --limit 5

# Force a re-pull even if files already exist
docker compose exec app python -m lofc.ingest.run --competition 2 --force

# Check what landed on disk
du -sh data/raw
find data/raw/2/27/events -type f | wc -l    # count of PL matches pulled so far
find data/raw -maxdepth 3 -type d            # folder layout


# ----- Phase 2: Aggregate -----
# Reads data/raw/, writes one row per player-season to data/processed/ (parquet + CSV).

# Aggregate all 3 leagues into the player-season metrics table
docker compose exec app python -m lofc.aggregate.run

# Aggregate one league only (2 = Premier League, 11 = La Liga, 12 = Serie A)
docker compose exec app python -m lofc.aggregate.run --competition 2

# Open the result (CSV is human-readable; parquet feeds the next phases)
open data/processed/player_season_metrics.csv          # opens in your spreadsheet app
docker compose exec app python -c "import pandas as pd; print(pd.read_parquet('data/processed/player_season_metrics.parquet').shape)"


# ----- Phase 3: Store -----
# Creates the database tables, loads the player table + the club reference data.

# Apply database migrations (creates the 4 tables)
docker compose exec app alembic upgrade head

# (Re)generate the constructed reference files in data/reference/
docker compose exec app python -m lofc.store.reference_data

# Load everything into Postgres (idempotent: safe to re-run, no duplicates)
docker compose exec app python -m lofc.store.load


# ----- Phase 4: Normalise & Score -----
# Ranks players vs positional peers (percentiles), builds performance + fit scores.

# Compute percentiles + scores and write them to Postgres (prints a spot-check)
docker compose exec app python -m lofc.model.run

# See the top centre-forwards by fit, or any position
docker compose exec db psql -U lofc -d lofc -c "SELECT m.player_name, s.performance_score, s.fit_score FROM player_scores s JOIN player_season_metrics m ON s.player_id=m.player_id AND s.competition_id=m.competition_id AND s.season_id=m.season_id WHERE s.competition_id=2 AND s.position_group='Centre Forward' ORDER BY s.fit_score DESC LIMIT 10;"

# ----- Phase 5: Archetypes -----
# Groups players by playing style within each position (k-means), labels each group.

# Cluster players into style archetypes and write them to Postgres
docker compose exec app python -m lofc.model.archetypes

# See the archetype labels and example players for a position
docker compose exec db psql -U lofc -d lofc -c "SELECT DISTINCT a.position_group, a.cluster_label FROM archetypes a WHERE a.position_group='Centre Forward';"


# ----- Phase 6: Valuation -----
# Estimates fair market value from performance and flags undervalued players.

# Download the Transfermarkt market-value data (one-time; gitignored, re-downloadable)
docker compose exec app python -m lofc.ingest.transfermarkt

# Train the valuation model and write fair value + undervaluation to Postgres
docker compose exec app python -m lofc.model.valuation

# See the biggest bargains (high fair value vs low market value), with names
docker compose exec db psql -U lofc -d lofc -c "SELECT m.player_name, m.team_name, v.position_group, v.age, v.market_value_eur, v.fair_value_eur, v.undervaluation_pct FROM valuations v JOIN player_season_metrics m USING(player_id,competition_id,season_id) WHERE m.minutes>1500 ORDER BY v.undervaluation_pct DESC LIMIT 10;"


# ----- Club scorecard: persist the 1-5 recruitment composite -----
# The live ranking model, written to player_scorecards so the offline shortlist and the BI
# layer read the same numbers as the dashboard. Runs AFTER valuation (Financial/Resale need
# it) and BEFORE the shortlist. Clear-then-insert, idempotent. Fast: no data fetch.

# Build/refresh the composite for every player-season (+ the Full Back / Winger archetypes)
docker compose exec app python -m lofc.model.scorecard_run

# See the top of a position on the club composite, with names
docker compose exec db psql -U lofc -d lofc -c "SELECT n.player_name, n.team_name, sc.objective_composite, sc.performance_band, sc.physical_band FROM player_scorecards sc JOIN player_metrics_neutral n USING(player_id,competition_id,season_id) WHERE sc.archetype='All Metrics' AND sc.season_id=318 AND sc.position_group='Centre Forward' ORDER BY sc.objective_composite DESC NULLS LAST LIMIT 10;"


# ----- Phase 7: Constrain & Rank -----
# Applies the affordability gates (fee + modelled wage) and ranks on the club's objective
# 1-5 composite (Performance + Physical). The retired Style-fit no longer ranks anything.

# Generate the default shortlists for every position and write them to Postgres
docker compose exec app python -m lofc.constrain.run

# See a position's shortlist (or near-misses), with names
docker compose exec db psql -U lofc -d lofc -c "SELECT s.rank, m.player_name, m.team_name, s.fit_score, s.market_value_eur, s.estimated_weekly_wage_gbp, s.wage_ceiling_gbp, s.is_near_miss FROM shortlists s JOIN player_season_metrics m USING(player_id,competition_id,season_id) WHERE s.position_group='Centre Forward' ORDER BY s.rank;"

pgrep cloudflared

# ----- Inspecting Postgres with SQL (now holds player + reference data) -----

# Open an interactive SQL shell (inside: \dt to list tables, \q to quit)
docker compose exec db psql -U lofc -d lofc

# One-liners (no interactive shell)
docker compose exec db psql -U lofc -d lofc -c "\dt"                       # list tables
docker compose exec db psql -U lofc -d lofc -c "SELECT count(*) FROM player_season_metrics;"
docker compose exec db psql -U lofc -d lofc -c "SELECT player_name, goals FROM player_season_metrics ORDER BY goals DESC LIMIT 5;"

# ----- Phase 10: Real data (paid StatsBomb + EFL Transfermarkt values) -----
# Targets come from SB_COMPETITIONS in .env (the 8 EFL league-seasons). Containers
# read .env at start, so re-up after changing it: docker compose up -d

# Verify which competitions/seasons the paid licence grants (read-only)
docker compose exec app python -c "from lofc.ingest import statsbomb; [print(c['competition_id'], c['competition_name'], c['season_name']) for c in statsbomb.get_competitions()]"

# Show the configured targets as the app resolves them
docker compose exec app python -c "from lofc.config import settings; [print(c.label) for c in settings.competitions]"

# Pull everything configured (resumable; ~4,450 matches on first run, takes hours)
docker compose exec app python -m lofc.ingest.run

# Pull one league only for a staged rollout (4 = League One)
docker compose exec app python -m lofc.ingest.run --competition 4

# Scrape current EFL market values from Transfermarkt squad pages (~100 requests,
# rate-limited, ~5 min; idempotent, --force to re-pull). The season is derived from
# today's date and printed at the start of the run. If contract/height/foot come back
# filled on under 20% of rows the run ABORTS without touching the CSV; on success the
# previous CSV is copied to data/backups/efl_values-<timestamp>.csv.
docker compose exec app python -m lofc.ingest.transfermarkt_efl

# Pin a specific Transfermarkt season (its starting year: 2026 = the 2026/27 season),
# or accept a pull that has lost a bio field. --allow-degraded is a deliberate override:
# it is what a bad scrape needs in order to overwrite good data, so use it knowingly.
docker compose exec app python -m lofc.ingest.transfermarkt_efl --force --season 2026
docker compose exec app python -m lofc.ingest.transfermarkt_efl --force --allow-degraded

# Reconcile the modelled wage grid against published club wage bills (report only)
docker compose exec app python -m lofc.model.wage_check

# Load the club-provided SkillCorner tracking export (xlsx in data/reference/skillcorner/)
# into Postgres: team-level = all 24 League One clubs, player-level = LOFC squad only.
docker compose exec app python -m lofc.ingest.skillcorner

# Peek at the recruiter watchlist (user data: survives every pipeline re-run)
docker compose exec db psql -U lofc -d lofc -c "SELECT w.status, w.note, p.player_name FROM watchlist w JOIN players p USING (player_id);"

# Sweep players with no metrics rows, keeping anyone on the watchlist (the watchlist
# foreign key blocks deleting watched players, by design)
docker compose exec db psql -U lofc -d lofc -c "DELETE FROM players WHERE player_id NOT IN (SELECT player_id FROM player_season_metrics) AND player_id NOT IN (SELECT player_id FROM watchlist);"

# ===== Phase 11: Impect (second provider, intended future primary) =====
# Credentials in .env: IMPECT_USERNAME / IMPECT_PASSWORD (re-up after editing .env).

# Pull player-season data for all configured Impect iterations (14 by default:
# the 4 EFL leagues + Scottish Premiership/Championship + PL2, 2024/25 + 2025/26).
# Idempotent: existing iteration files are skipped; --force re-pulls; --iteration N
# pulls one only. Lands parquet under data/raw/impect/ (gitignored, licensed data).
# IN-SEASON: the live season (config.LIVE_SEASON_ID) is ALWAYS re-pulled, because Impect
# returns season-to-date aggregates -- a landed file goes stale after every matchweek.
# Finished seasons stay skip-if-exists. A season with no matches yet is skipped, not an error.
docker compose exec app python -m lofc.ingest.impect

# ----- WEEKLY IN-SEASON REFRESH (run after each round of fixtures) -----
# Both providers always re-pull the live season (config.LIVE_SEASON_ID); finished seasons and
# not-yet-started leagues are skipped. 1. events  2. physical  3. metric table  4. models.
docker compose exec app python -m lofc.ingest.impect
docker compose exec app python -m lofc.ingest.skillcorner_api
docker compose exec app python -m lofc.model.build_neutral --write
docker compose exec app python -m lofc.pipeline

# Validation gate (read-only, no DB writes): match Impect players to ours (TM id,
# then birth date + name) and compare per-player goal totals + xG against the
# StatsBomb data we already validated. Must look healthy for the EFL before any
# downstream build on Impect.
docker compose exec app python -m lofc.model.impect_check

# ===== SkillCorner platform (physical data for ALL players in covered leagues) =====
# Replaces the squad-only xlsx as the physical source. Credentials in .env:
# SKILLCORNER_USERNAME / SKILLCORNER_PASSWORD (re-up after editing .env).

# Pull season physical data for all covered editions (7 by default: the 4 EFL
# leagues + PL2 + Scottish Premiership + Irish Premier, 2025/26). One row per player,
# season-aggregated. Idempotent; --force re-pulls; --edition N pulls one.
# Lands parquet under data/raw/skillcorner/ (gitignored, licensed data).
docker compose exec app python -m lofc.ingest.skillcorner_api

# Coverage report (read-only): how many SkillCorner players tie to a player in our DB.
# High rate for the EFL editions = physical data now available for those candidates.
docker compose exec app python -m lofc.model.skillcorner_check

# ===== StatsBomb Season Player Stats (only the metrics Impect lacks) =====
# Pulls ready-made per-90 advanced metrics (PAdj tackles/interceptions + pressures,
# aggressive actions, pressures in opp half, xG buildup, pressured-pass %, box-cross %)
# for the 8 EFL league-seasons. player_id = our player_id, so no matching. Idempotent.
# Lands parquet under data/raw/statsbomb_season/ (gitignored, licensed data).
docker compose exec app python -m lofc.ingest.statsbomb_season

# ===== Unified metric registry + neutral-layer merge (Phase 11 Steps 1) =====
# Print the full 87-metric inventory: every metric, its single source, and the exact
# derivation (registry invariants are asserted on import).
docker compose exec app python -m lofc.model.metric_registry

# Read-only merge report: combines Impect + StatsBomb (advanced + computed) + SkillCorner
# into one wide table per player for every configured league season, and prints per-source
# coverage. Writes nothing to the database.
docker compose exec app python -m lofc.model.build_neutral

# Load the combined 87-metric table into the database (player_metrics_neutral,
# clear-then-insert per league season; also pipeline step 15). Without --write = report only.
docker compose exec app python -m lofc.model.build_neutral --write

# ===== Injury history (Medical dimension input) =====
# Pull Transfermarkt injury histories for EFL players (~1.8 hours, resumable).
docker compose exec app python -m lofc.ingest.transfermarkt_injuries --force
# Load the scraped injury history into Postgres (fast; also runs inside the pipeline).
docker compose exec app python -m lofc.store.injuries
